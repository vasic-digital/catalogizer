package ftp

import (
	"bufio"
	"context"
	"io"
	"net"
	"testing"
	"time"

	"github.com/stretchr/testify/require"
)

type logConn struct {
	net.Conn
	t *testing.T
}

func (l *logConn) Read(b []byte) (int, error) {
	n, err := l.Conn.Read(b)
	if n > 0 {
		l.t.Logf("  <- %q", string(b[:n]))
	}
	return n, err
}
func (l *logConn) Write(b []byte) (int, error) {
	l.t.Logf("  -> %q", string(b))
	return l.Conn.Write(b)
}

func TestZZDebug_RawTraffic(t *testing.T) {
	f := load(t)
	for _, mode := range []string{"tls", "clear"} {
		cfg := f.cfg(nil)
		if mode == "clear" {
			cfg.TLSMode, cfg.TrustedLAN = TLSNone, true
		} else {
			cfg.PinStore = f.pinned(t)
		}
		cfg.IOTimeout = 4 * time.Second
		c := NewFTPClient(cfg)
		require.NoError(t, c.Connect(ictx(t)))
		require.NoError(t, c.acquire(ictx(t)))
		p := c.p
		p.ctl = &logConn{Conn: p.ctl, t: t}
		p.br = bufio.NewReaderSize(p.ctl, 4096)
		ctx := ictx(t)
		t.Logf("== %s: RETR, read 10, close data", mode)
		dc, err := p.beginTransfer(ctx, 0, "RETR %s", "/sub/b.bin")
		require.NoError(t, err)
		_, err = io.ReadFull(dc, make([]byte, 10))
		require.NoError(t, err)
		_ = dc.Close()
		code, lines, err := p.reply(ctx, "transfer", nil)
		t.Logf("reply: %d %q %v", code, lines, err)
		code, lines, err = p.cmd(ctx, nil, "NOOP")
		t.Logf("noop: %d %q %v", code, lines, err)
		t.Logf("== %s: full RETR of a.txt", mode)
		dc, err = p.beginTransfer(ctx, 0, "RETR %s", "/a.txt")
		require.NoError(t, err)
		_, _ = io.Copy(io.Discard, dc)
		_ = dc.Close()
		code, lines, err = p.reply(ctx, "transfer", nil)
		t.Logf("reply: %d %q %v", code, lines, err)
		c.release()
		_ = c.Disconnect(context.Background())
	}
}
