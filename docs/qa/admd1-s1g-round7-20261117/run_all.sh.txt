cd $W/catalog-api || exit 1

export GOMAXPROCS=2
RE='ADMD1|Admin403|ContractGate|OpenAPI|R6|R7|S1G6'
for i in 1 2; do
  { echo "# commit-base $(git -C $W rev-parse HEAD) green run $i"; nice -n 10 go test -vet=off -p 1 -count=1 -run "$RE" . 2>&1 | grep -v '^\[GIN'; echo "exit=${PIPESTATUS[0]}"; } > $E/green_$i.txt
done
{ nice -n 10 go vet . 2>&1 | tail -5; echo "vet exit=${PIPESTATUS[0]}"; gofmt -l api_contract_gate_s1g6_test.go api_contract_gate_groundtruth_test.go api_contract_gate_codescan_test.go; echo "gofmt done"; } > $E/vet_fmt.txt

run() { # name expect file old new tests
  local name=$1 expect=$2 file=$3 old=$4 new=$5 tests=$6
  python3 - "$file" "$old" "$new" <<'PY'
import sys
f,o,n=sys.argv[1:4]
s=open(f).read()
assert o in s, "pattern missing"
open(f,'w').write(s.replace(o,n,1))
PY
  if [ $? -ne 0 ]; then echo "$name: MUTATION NOT APPLIED" >> $O; return; fi
  local res; res=$(nice -n 10 go test -vet=off -p 1 -count=1 -run "$tests" . 2>&1 | grep -v '^\[GIN')
  local v=PASS; echo "$res" | grep -q '^FAIL' && v=FAIL
  local ok=UNEXPECTED; [ "$v" = "$expect" ] && ok=OK
  { echo "## $name | $file | tests=$tests | expected $expect | got $v | $ok"; echo "$res" | grep -E 'Error:|route-level|handler-wrapping|not used|differs|--- FAIL|^ok|^FAIL' | cut -c1-240 | head -6; } >> $O
  cp $W/.main.orig main.go; cp $W/.gt.orig api_contract_gate_groundtruth_test.go; cp $W/.table.orig ../docs/ROUTE_AUTHZ_TABLE.md
}
G='api.POST("/admin/covers/generate-thumbnails", jwtMiddleware.RequireAdmin(), '
