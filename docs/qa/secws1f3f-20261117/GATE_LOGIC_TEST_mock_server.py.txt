import http.server, socketserver, struct, sys, os, time, threading, hashlib, base64, json

PORT, MODE, TRIG = int(sys.argv[1]), sys.argv[2], sys.argv[3]
USED = set(); EXPS = {}; LOCK = threading.Lock(); TKUSER = {}

def b64(d): return base64.urlsafe_b64encode(json.dumps(d).encode()).decode().rstrip("=")
def jwt(user):
    exp = int(time.time()) + (12 if MODE == "short-exp" else (22 if (MODE == "expiry-race" and user == "c") else (90 if (MODE == "expiry-race-long" and user == "c") else 36000))); EXPS[user] = exp
    return b64({"alg": "none"}) + "." + b64({"sub": user, "exp": exp}) + ".sig"
def frame(op, payload=b""): return bytes([0x80 | op, len(payload)]) + payload
def suspended(user): return user == "c" and os.path.exists(TRIG) and MODE != "ignore-suspension"

class H(http.server.BaseHTTPRequestHandler):
    protocol_version = "HTTP/1.1"
    def log_message(self, *a): pass
    def handle_one_request(self):

        except (ConnectionError, OSError): self.close_connection = True
    def _json(self, body, code=200):
        b = body.encode(); self.send_response(code); self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(b))); self.end_headers(); self.wfile.write(b)
    def do_POST(self):

        if self.path.endswith("/auth/login"):
            u = json.loads(body or b"{}").get("username", "")
            if suspended(u): return self._json('{"error":"suspended"}', 403)
            self._json(json.dumps({"token": jwt(u)}))
        elif self.path.endswith("/ws/ticket"):

            try: p = tok.split(".")[1]; u = json.loads(base64.urlsafe_b64decode(p + "=" * (-len(p) % 4)))["sub"]
            except Exception: return self._json("{}", 401)

            t = "t%s_%d" % (u, int(time.time() * 1e6)); TKUSER[t] = u; self._json('{"ticket":"%s"}' % t)
        else: self._json("{}", 404)

        t = q[7:]

            USED.add(t)
        user = TKUSER[t]
        key = self.headers["Sec-WebSocket-Key"]
        acc = base64.b64encode(hashlib.sha1((key + "258EAFA5-E914-47DA-95CA-C5AB0DC85B11").encode()).digest()).decode()
        self.wfile.write(("HTTP/1.1 101 Switching Protocols\r\nUpgrade: websocket\r\nConnection: Upgrade\r\n"

        t0 = time.time(); trig_seen = None

        while True:
