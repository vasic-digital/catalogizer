run(){ # scenario mockmode want_exit want_f3 suspend_cmd [extra env...]
  [ "${NEW_ONLY:-0}" = 1 ] && [ -z "${R7:-}" ] && return 0   # NEW_ONLY: skip the pre-round-7 scenarios (RED demonstrations)

  local trig="$S/trig_$name"; rm -f "$trig"
  local sp up=0 try alive
  for try in 1 2 3; do
    python3 -I "$HERE/GATE_LOGIC_TEST_mock_server.py.txt" "$PORT" "$mode" "$trig" 2> "$S/mock_$name.err" & sp=$!
    for _ in $(seq 1 40); do
      kill -0 "$sp" 2>/dev/null || break
      curl -s --max-time 1 -o /dev/null "http://127.0.0.1:$PORT/health" && { up=1; break; }; sleep 0.25
    done
    [ "$up" = 1 ] && break; kill "$sp" 2>/dev/null; wait "$sp" 2>/dev/null; PORT=$((PORT+1))
  done
  expect "scenario:$name mock up (health answered)" "$up" 1
  env BASE="http://127.0.0.1:$PORT" ADMIN_USER=a ADMIN_PASS=b VICTIM_USER=c VICTIM_PASS=d OUT="$S/out_$name" \
    SUSPEND_CMD="${susp//TRIG/$trig}" F3_BOUND="${F3_BOUND:-6}" "$@" bash "$SCRIPT" > "$S/log_$name.txt" 2>&1 < /dev/null
  local rc=$?; if kill -0 "$sp" 2>/dev/null; then alive=1; else alive=0; fi; kill "$sp" 2>/dev/null; wait "$sp" 2>/dev/null
  expect "scenario:$name mock still alive at end" "$alive" 1
  local f3; f3="$(grep '^[A-Z]* | F3 socket closed because' "$S/out_$name/results.txt" 2>/dev/null | head -1 | cut -d' ' -f1)"
  expect "scenario:$name exit" "$rc" "$wexit"; expect "scenario:$name F3 verdict" "${f3:-NONE}" "$wf3"
}
