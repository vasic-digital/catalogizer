#!/usr/bin/env bash
# GATE-LOGIC UNIT TEST of LIVE_WS_CHECK.sh.txt (11.4.289(C)). NOT live evidence: the target is a local mock
# (GATE_LOGIC_TEST_mock_server.py.txt). It only proves the script's verdict logic on declared scenarios.
# Round 6 (R5-1..R5-8). usage: bash GATE_LOGIC_TEST_run.sh.txt <scratch-dir>   (env SCRIPT=<path> runs the suite against another script copy, e.g. the round-4 script to show RED)
# Each scenario asserts BOTH the overall exit code and the F3 line verdict. Overall exit is 3 whenever the F3 check is PASS
# (the audience leg is always INCONCLUSIVE: uncovered) and 1 on an F3 FAIL; the F3 PASS must appear ONLY in scenario 'good'.
set -u
HERE="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"; S="${1:?scratch dir}"; mkdir -p "$S"; SCRIPT="${SCRIPT:-$HERE/LIVE_WS_CHECK.sh.txt}"
PORT=$((20000 + RANDOM % 20000)); fails=0
expect(){ if [ "$2" = "$3" ]; then echo "UNIT-PASS $1 ($2)"; else echo "UNIT-FAIL $1 (got $2, expected $3)"; fails=$((fails+1)); fi; }
run(){ # scenario mockmode want_exit want_f3 suspend_cmd [extra env...]
  local name="$1" mode="$2" wexit="$3" wf3="$4" susp="$5"; shift 5; PORT=$((PORT+1))
  local trig="$S/trig_$name"; rm -f "$trig"
  local sp up=0 try alive
  for try in 1 2 3; do
    python3 -I "$HERE/GATE_LOGIC_TEST_mock_server.py.txt" "$PORT" "$mode" "$trig" 2> "$S/mock_$name.err" & sp=$!
    for _ in $(seq 1 40); do
      kill -0 "$sp" 2>/dev/null || break
      curl -s --max-time 1 -o /dev/null "http://127.0.0.1:$PORT/health" && { up=1; break; }; sleep 0.25
    done
    [ "$up" = 1 ] && break; kill "$sp" 2>/dev/null; wait "$sp" 2>/dev/null; PORT=$((PORT+1))
  done
  expect "scenario:$name mock up (health answered)" "$up" 1
  env BASE="http://127.0.0.1:$PORT" ADMIN_USER=a ADMIN_PASS=b VICTIM_USER=c VICTIM_PASS=d OUT="$S/out_$name" \
    SUSPEND_CMD="${susp//TRIG/$trig}" F3_BOUND="${F3_BOUND:-6}" "$@" bash "$SCRIPT" > "$S/log_$name.txt" 2>&1 < /dev/null
  local rc=$?; if kill -0 "$sp" 2>/dev/null; then alive=1; else alive=0; fi; kill "$sp" 2>/dev/null; wait "$sp" 2>/dev/null
  expect "scenario:$name mock still alive at end" "$alive" 1
  local f3; f3="$(grep 'F3 socket closed because' "$S/out_$name/results.txt" 2>/dev/null | head -1 | cut -d' ' -f1)"
  expect "scenario:$name exit" "$rc" "$wexit"; expect "scenario:$name F3 verdict" "${f3:-NONE}" "$wf3"
}
# 1. helper functions (source-only)
export LIVE_WS_CHECK_SOURCE_ONLY=1; BASE=x OUT="$S"; . "$SCRIPT"; unset LIVE_WS_CHECK_SOURCE_ONLY
[ "$(hostonly https://h.example:8443/x)" = h.example ] && echo "UNIT-PASS hostonly strips :port" || { echo "UNIT-FAIL hostonly"; fails=$((fails+1)); }
# code el bound end ctrl susp exp => verdict
for c in "1008 3.0 20 close pong ok ok PASS" "1008 25 20 close pong ok ok FAIL" "1000 3 20 close pong ok ok FAIL" \
         "null 0 20 timeout pong ok ok FAIL" "null 0 20 stopped pong ok ok FAIL" "null 0 20 eof pong ok ok INCONCLUSIVE" \
         "1008 -2 20 close pong ok ok INCONCLUSIVE" "1008 3 20 close closed/final_pong=false ok ok INCONCLUSIVE" \
         "1008 3 20 close pong none ok INCONCLUSIVE" "1008 3 20 close pong ok none INCONCLUSIVE"; do
  set -- $c; r="$(f3_verdict "$1" "$2" "$3" "$4" "$5" "$6" "$7")"
  [ "${r%%|*}" = "$8" ] && echo "UNIT-PASS f3_verdict $c" || { echo "UNIT-FAIL f3_verdict $c => $r"; fails=$((fails+1)); }
done
# 2. whole-script scenarios against the mock
run good           good              3 PASS         'touch TRIG'
run wrong-code     wrong-code        1 FAIL         'touch TRIG'
run never-close    never-close       1 FAIL         'touch TRIG' F3_BOUND=3
run refuse-ws      refuse-ws         3 INCONCLUSIVE 'touch TRIG'
run drop-after-101 drop-after-101    3 INCONCLUSIVE 'touch TRIG'
run early-close    early-close       3 INCONCLUSIVE 'sleep 2; touch TRIG'
run no-pong        no-pong           3 INCONCLUSIVE 'touch TRIG'
run late-close     late-close        1 FAIL         'touch TRIG' F3_BOUND=3
run suspend-fails  never-close       3 INCONCLUSIVE 'false'
run ignore-susp    ignore-suspension 3 INCONCLUSIVE 'touch TRIG'
run control-closes control-closes-too 3 INCONCLUSIVE 'touch TRIG'
run short-exp      short-exp         3 INCONCLUSIVE 'touch TRIG'
run control-later  control-later     3 INCONCLUSIVE 'touch TRIG'
# 14. victim probe SIGKILLed mid-run (no JSON), victim socket still OPEN when killed (mock never-close, so a missing-JSON-as-timeout
#     mutant reads FAIL). The wrapper kills the victim's real probe after 2 s and exec()s the real probe directly for every other probe.
cat > "$S/killprobe2.py" <<'PYEOF'
import os, subprocess, sys, time
real = os.environ["REAL_PROBE"]; args = sys.argv[1:]
if not ("hold" in args and os.environ.get("WS_TICKET", "").startswith("tc_")):
    os.execv(sys.executable, [sys.executable, "-I", real] + args)
p = subprocess.Popen([sys.executable, "-I", real] + args)
time.sleep(2); p.kill(); p.wait(); sys.exit(137)
PYEOF
run probe-killed2  never-close       3 INCONCLUSIVE 'touch TRIG' PROBE="$S/killprobe2.py" REAL_PROBE="$HERE/ws_probe.py.txt"
# 16. N1 expiry race: token valid at start, expires mid-SUSPEND_CMD; margin 5 so a ~22 s mock token passes the start check. Marker at ~18 s: ~4 s left < bound+margin.
run expiry-race    expiry-race       3 INCONCLUSIVE 'sleep 12; touch TRIG' F3_BOUND=6 F3_EXP_MARGIN=5
# 16b. SLOW (RUN_SLOW=1): the reviewer's round-4 race with DEFAULT margin 75: token 90 s, marker at ~87 s. Against the round-4 script this reads PASS (RED).
if [ "${RUN_SLOW:-0}" = 1 ]; then run expiry-race-long expiry-race-long 3 INCONCLUSIVE 'sleep 84; touch TRIG' F3_BOUND=6; fi
# 17. N2 late close: SUSPEND_CMD returns 5 s after its effect; close 8 s after the effect = 3 s after return (<= bound 6) but 8 s after start (> bound)
run late-return    late-close        3 INCONCLUSIVE 'touch TRIG; sleep 5' F3_BOUND=6
# 18. N1 timeout: SUSPEND_CMD overruns SUSPEND_TIMEOUT
run suspend-timeout never-close      3 INCONCLUSIVE 'sleep 30' SUSPEND_TIMEOUT=3
# 19. N3 injection: forged verdict line in the server close reason. Exactly ONE line may mention the F3 check and none may start with PASS.
run inject         inject            1 FAIL         'touch TRIG'
n="$(grep -c 'F3 socket closed because' "$S/out_inject/results.txt")"; expect "inject: exactly one F3 verdict line" "$n" 1
p="$(grep -c '^PASS | F3' "$S/out_inject/results.txt")"; expect "inject: no forged PASS line" "$p" 0
# 21. R5-4 N1-ONLY: token ~22 s, margin 5, bound 20 -> need 25 > remaining: INCONCLUSIVE; the close at expiry is <= bound from BOTH the command
#     start and return, so ONLY the N1 lifetime check prevents a PASS (the M-N1-only mutant below reads PASS)
run n1-only        expiry-race       3 INCONCLUSIVE 'touch TRIG' F3_BOUND=20 F3_EXP_MARGIN=5
# 22. R5-1: bad numeric settings exit 2 at startup (round-5 script: arithmetic error, rc 0, all PASS, no F3 line)
BADENV=(BASE=http://127.0.0.1:1 ADMIN_USER=a ADMIN_PASS=b VICTIM_USER=c VICTIM_PASS=d)
for bad in "F3_BOUND=20.5" "F3_BOUND=-1" "F3_BOUND=" "F3_EXP_MARGIN=abc" "SUSPEND_TIMEOUT=1e3" "F3_BOUND=1234567890"; do
  env "${BADENV[@]}" OUT="$S/out_bad" "$bad" bash "$SCRIPT" >"$S/log_bad.txt" 2>&1 </dev/null; rcb=$?
  expect "R5-1 $bad exit" "$rcb" 2
done
env "${BADENV[@]}" OUT="$S/out_lz" F3_BOUND=08 bash "$SCRIPT" >"$S/log_lz.txt" 2>&1 </dev/null; rcl=$?
[ "$rcl" != 2 ] && echo "UNIT-PASS R5-1 leading zero 08 accepted as decimal" || { echo "UNIT-FAIL R5-1 leading zero rejected"; fails=$((fails+1)); }
# 23. R5-1 end assertion: login impossible (closed port) -> BOTH expected F3 lines reported absent, exit 3
env "${BADENV[@]}" OUT="$S/out_nologin" bash "$SCRIPT" >"$S/log_nologin.txt" 2>&1 </dev/null; rcn=$?; expect "R5-1 no-login exit" "$rcn" 3
n="$(grep -c 'expected F3 result line absent' "$S/out_nologin/results.txt")"; expect "R5-1 two absent-line records" "$n" 2
# 24. R5-2: SUSPEND_CMD stdin is /dev/null even when the script runs on a tty (python pty): a command that tests -t 0 must see no tty
cat > "$S/ptyrun.py" <<'PYEOF'
import os, pty, sys, time
pid, fd = pty.fork()
if pid == 0:
    os.execvp("bash", ["bash"] + sys.argv[1:])
t0 = time.time(); buf = b""
while True:
    try: d = os.read(fd, 4096)
    except OSError: break
    if not d: break
    buf += d
_, st = os.waitpid(pid, 0); print("elapsed=%.1f rc=%d" % (time.time() - t0, os.waitstatus_to_exitcode(st)))
PYEOF
mockup(){ # mode trigfile -> sets PORT, sp
  PORT=$((PORT+1)); python3 -I "$HERE/GATE_LOGIC_TEST_mock_server.py.txt" "$PORT" "$1" "$2" 2>/dev/null & sp=$!
  for _ in $(seq 1 40); do curl -s --max-time 1 -o /dev/null "http://127.0.0.1:$PORT/health" && break; sleep 0.25; done
}
trig="$S/trig_pty"; rm -f "$trig"; mockup good "$trig"
env "${BADENV[@]}" BASE="http://127.0.0.1:$PORT" OUT="$S/out_pty" F3_BOUND=6 SUSPEND_CMD="test -t 0 && exit 7; touch $trig" python3 -I "$S/ptyrun.py" "$SCRIPT" > "$S/log_pty.txt" 2>&1
kill "$sp" 2>/dev/null; wait "$sp" 2>/dev/null
f3="$(grep 'F3 socket closed because' "$S/out_pty/results.txt" 2>/dev/null | head -1 | cut -d' ' -f1)"; expect "R5-2 pty: SUSPEND_CMD has no tty stdin (F3 verdict)" "${f3:-NONE}" PASS
# 25. R5-8: SIGTERM during SUSPEND_CMD leaves no probe behind and exits 143
trig="$S/trig_term"; rm -f "$trig"; mockup never-close "$trig"
env "${BADENV[@]}" BASE="http://127.0.0.1:$PORT" OUT="$S/out_term" F3_BOUND=6 SUSPEND_CMD='exec sleep 60' bash "$SCRIPT" > "$S/log_term.txt" 2>&1 < /dev/null & bp=$!
for _ in $(seq 1 40); do grep -q 'victim and control sockets alive' "$S/log_term.txt" 2>/dev/null && break; sleep 0.5; done
sleep 1; kill -TERM "$bp"; wait "$bp"; rct=$?; expect "R5-8 SIGTERM exit status" "$rct" 143; sleep 1
left="$(pgrep -f "[w]s_probe.py.txt ws://127.0.0.1:$PORT" | wc -l)"; expect "R5-8 no probe left after SIGTERM" "$left" 0
kill "$sp" 2>/dev/null; wait "$sp" 2>/dev/null
# 26. R5-5 sanitizer: C1 (U+0085/U+009B), U+2028/2029, DEL, newline, pipe, invalid UTF-8, multibyte truncation
s="$(san $'a\xc2\x85b\xe2\x80\xa8c\xe2\x80\xa9d\x7fe\nf|g\xc2\x9bh')"; expect "R5-5 san strips C1/LS/PS/DEL/NL/pipe" "$s" "abcdef/gh"
n="$(san $'x\xc2\x85PASS | F3 forged' | python3 -I -c 'import sys; print(len(sys.stdin.read().splitlines()))')"; expect "R5-5 san output is one logical line" "$n" 1
long="$(python3 -I -c 'print("é" * 300)')"; n="$(san "$long" | python3 -I -c 'import sys; print(len(sys.stdin.buffer.read().decode("utf-8")))')"; expect "R5-5 truncation by character, valid UTF-8" "$n" 200
s="$(san $'ok\xff\xfe')"; [ "${#s}" -ge 2 ] && echo "UNIT-PASS R5-5 san survives invalid UTF-8" || { echo "UNIT-FAIL R5-5 invalid UTF-8"; fails=$((fails+1)); }
# 15. usage / output-directory errors must exit 2 (not 1)
set +e
env -u BASE ADMIN_USER=a ADMIN_PASS=b VICTIM_USER=c VICTIM_PASS=d OUT="$S/out_nobase" bash "$SCRIPT" >/dev/null 2>&1 </dev/null; expect "usage: BASE unset exit" "$?" 2
BASE=http://127.0.0.1:1 ADMIN_USER=a ADMIN_PASS=b VICTIM_USER="" VICTIM_PASS=d OUT="$S/out_novu" bash "$SCRIPT" >/dev/null 2>&1 </dev/null; expect "usage: VICTIM_USER empty exit" "$?" 2
: > "$S/afile"
BASE=http://127.0.0.1:1 ADMIN_USER=a ADMIN_PASS=b VICTIM_USER=c VICTIM_PASS=d OUT="$S/afile/sub" bash "$SCRIPT" >"$S/log_unwritable.txt" 2>&1 </dev/null; expect "unwritable OUT exit" "$?" 2
grep -q 'cannot create output directory' "$S/log_unwritable.txt" && echo "UNIT-PASS unwritable OUT message" || { echo "UNIT-FAIL unwritable OUT message"; fails=$((fails+1)); }
# 20. NAMED MUTANTS of the script (each must make its target scenario give a WRONG verdict, i.e. the suite detects it)
mutate(){ # name old new file : python string replacement, asserts exactly one occurrence
  python3 -I - "$1" "$2" "$3" "$SCRIPT" "$S/mut_$1.sh" <<'PYEOF'
import sys
n, old, new, src, dst = sys.argv[1:6]
t = open(src).read(); assert t.count(old) == 1, (n, t.count(old)); open(dst, "w").write(t.replace(old, new))
PYEOF
}
mutscen(){ # mutantname mockmode wexit wf3 suspend [env...]  (want values = what the MUTANT wrongly gives)
  local m="$1"; shift; PROBE="$HERE/ws_probe.py.txt" SCRIPT="$S/mut_$m.sh" run "mut_$m" "$@"
}
# N4: a missing victim JSON read as "still open" turns the killed-probe case into a false FAIL
mutate M-N4-noprobe-as-timeout 'then END=noprobe; CC=""; fi' 'then :; fi; [ -z "$END" ] && END=timeout' && \
  mutscen M-N4-noprobe-as-timeout never-close 1 FAIL 'touch TRIG' PROBE="$S/killprobe2.py" REAL_PROBE="$HERE/ws_probe.py.txt"
# N1+N2 together (the expiry race is caught by BOTH: close at the expiry is also > bound after the command START; honest redundancy)
mutate M-N1-expiry-at-start 'rem="$(jwt_exp_remaining "$VT")"   # N1' 'rem=9999   # N1' && \
  cp "$S/mut_M-N1-expiry-at-start.sh" "$S/mut_M-N1N2.sh" && SCRIPT="$S/mut_M-N1N2.sh" mutate M-N1N2 'elst="${8:-$2}"' 'elst=0' && \
  mutscen M-N1N2 expiry-race 3 PASS 'sleep 14; touch TRIG' F3_BOUND=9 F3_EXP_MARGIN=5
mutate M-N2-no-start-check 'elst="${8:-$2}"' 'elst=0' && mutscen M-N2-no-start-check late-close 3 PASS 'touch TRIG; sleep 5' F3_BOUND=6
# R5-4: the N1 lifetime re-check alone (N2 left intact) is the only thing separating n1-only from PASS
mutate M-N1-only 'rem="$(jwt_exp_remaining "$VT")"   # N1' 'rem=9999   # N1' && mutscen M-N1-only expiry-race 3 PASS 'touch TRIG' F3_BOUND=20 F3_EXP_MARGIN=5
# R5-1: validation removed -> a bad F3_BOUND no longer exits 2 (validation + normalisation both removed, so 20.5 is not rejected)
mutate M-R51-novalidate1 '  case "${!v}" in' '  case "0" in' && cp "$S/mut_M-R51-novalidate1.sh" "$S/mut_M-R51-novalidate.sh" && SCRIPT="$S/mut_M-R51-novalidate.sh" mutate M-R51-novalidate2 "printf -v \"\$v\" '%s' \"\$((10#\$val))\"" ':' && { env "${BADENV[@]}" PROBE="$HERE/ws_probe.py.txt" OUT="$S/out_mv" F3_BOUND=20.5 bash "$S/mut_M-R51-novalidate2.sh" >/dev/null 2>&1 </dev/null; rcm=$?; expect "mutant M-R51-novalidate: bad value no longer exits 2 (suite detects it)" "$([ "$rcm" = 2 ] && echo 2 || echo not2)" not2; }
# R5-1: end assertion removed -> no absent-line records
mutate M-R51-noendcheck "for pat in 'F3 socket closed because of suspension' 'F3 audience filter'; do" 'for pat in; do' && { env "${BADENV[@]}" OUT="$S/out_me" PROBE="$HERE/ws_probe.py.txt" bash "$S/mut_M-R51-noendcheck.sh" >/dev/null 2>&1 </dev/null; n="$(grep -c 'expected F3 result line absent' "$S/out_me/results.txt")"; expect "mutant M-R51-noendcheck: absent lines no longer recorded (suite detects it)" "$n" 0; }
# R5-5: weaker sanitizer lets U+0085 through
mutate M-R55-oldsan 'unicodedata.category(c) not in ("Cc", "Cf", "Zl", "Zp")' 'ord(c) >= 32 and ord(c) != 127' && { s="$( ( LIVE_WS_CHECK_SOURCE_ONLY=1; . "$S/mut_M-R55-oldsan.sh"; san $'a\xc2\x85b' ) )"; [ "$s" = $'a\xc2\x85b' ] && echo "UNIT-PASS mutant M-R55-oldsan: U+0085 survives (suite detects it)" || { echo "UNIT-FAIL M-R55 mutant not detected"; fails=$((fails+1)); }; }
# N3: sanitizer removed => forged verdict line lands in results.txt
mutate M-N3-no-sanitize '"/" if c == "|" else c for c in s if unicodedata.category(c) not in ("Cc", "Cf", "Zl", "Zp")' 'c for c in s' && mutscen M-N3-no-sanitize inject 1 FAIL 'touch TRIG'
n="$(grep -c '^PASS | F3' "$S/out_mut_M-N3-no-sanitize/results.txt")"; expect "mutant M-N3: forged PASS line appears (suite detects the mutant)" "$n" 1
for n in good ignore-susp control-closes; do echo "--- $n results.txt ---"; cat "$S/out_$n/results.txt"; done
echo "gate-logic failures: $fails"; exit "$fails"
