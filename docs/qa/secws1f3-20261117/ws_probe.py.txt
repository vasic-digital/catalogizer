#!/usr/bin/env python3
# Minimal stdlib RFC6455 client used by LIVE_WS_CHECK.sh.txt (no websocat needed).
# Modes: handshake (prints HTTP status, closes) | hold (after 101: ping/pong liveness proof, then read until
# close frame / EOF / --max-seconds; reports close code, reason and epoch). Last stdout line is one JSON object.
# Env INSECURE=1 disables TLS verification (test hosts only). Never prints credentials; URL query is not echoed.
import argparse, base64, json, os, socket, ssl, struct, time
from urllib.parse import urlsplit

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("url"); ap.add_argument("--origin", default="")
    ap.add_argument("--mode", choices=["handshake", "hold"], default="handshake")
    ap.add_argument("--max-seconds", type=float, default=60)
    ap.add_argument("--ready-file", default="")
    a = ap.parse_args()
    res = {"handshake": 0, "alive": False, "close_code": None, "close_reason": None,
           "close_epoch": None, "end": "error", "frames": 0}
    try:
        u = urlsplit(a.url); tls = u.scheme in ("wss", "https")
        port = u.port or (443 if tls else 80)
        s = socket.create_connection((u.hostname, port), timeout=10)
        if tls:
            ctx = ssl.create_default_context()
            if os.environ.get("INSECURE") == "1":
                ctx.check_hostname = False; ctx.verify_mode = ssl.CERT_NONE
            s = ctx.wrap_socket(s, server_hostname=u.hostname)
        path = (u.path or "/") + ("?" + u.query if u.query else "")
        key = base64.b64encode(os.urandom(16)).decode()
        req = ("GET %s HTTP/1.1\r\nHost: %s\r\nConnection: Upgrade\r\nUpgrade: websocket\r\n"
               "Sec-WebSocket-Version: 13\r\nSec-WebSocket-Key: %s\r\n" % (path, u.netloc, key))
        if a.origin: req += "Origin: %s\r\n" % a.origin
        s.sendall((req + "\r\n").encode())
        buf = b""; s.settimeout(10)
        while b"\r\n\r\n" not in buf:
            c = s.recv(4096)
            if not c: break
            buf += c
        head, _, buf = buf.partition(b"\r\n\r\n")
        try: res["handshake"] = int(head.split(b" ", 2)[1])
        except Exception: res["handshake"] = 0
        if res["handshake"] != 101 or a.mode == "handshake":
            res["end"] = "handshake-only" if res["handshake"] == 101 else "refused"
            return res
        def send(op, payload=b""):
            m = os.urandom(4)
            s.sendall(bytes([0x80 | op, 0x80 | len(payload)]) + m + bytes(b ^ m[i % 4] for i, b in enumerate(payload)))
        send(9, b"p")  # ping: a pong proves the server side is alive
        deadline = time.time() + a.max_seconds; s.settimeout(0.5)
        while time.time() < deadline:
            while len(buf) >= 2:  # parse complete frames from buffer
                op = buf[0] & 0x0F; ln = buf[1] & 0x7F; off = 2
                if ln == 126:
                    if len(buf) < 4: break
                    ln = struct.unpack(">H", buf[2:4])[0]; off = 4
                elif ln == 127:
                    if len(buf) < 10: break
                    ln = struct.unpack(">Q", buf[2:10])[0]; off = 10
                if len(buf) < off + ln: break
                pl = buf[off:off + ln]; buf = buf[off + ln:]; res["frames"] += 1
                if op == 10:
                    res["alive"] = True
                    if a.ready_file: open(a.ready_file, "w").write(str(time.time()))
                elif op == 9: send(10, pl)
                elif op == 8:
                    res["close_epoch"] = time.time(); res["end"] = "close"
                    if len(pl) >= 2: res["close_code"] = struct.unpack(">H", pl[:2])[0]
                    res["close_reason"] = pl[2:].decode("utf-8", "replace")
                    return res
            try:
                c = s.recv(4096)
            except socket.timeout:
                continue
            if not c:
                res["close_epoch"] = time.time(); res["end"] = "eof"; return res
            buf += c
        res["end"] = "timeout"
    except Exception as e:
        res["end"] = "error:" + type(e).__name__
    return res

if __name__ == "__main__":
    print(json.dumps(main()))
