import http.server, socketserver, struct, sys, os, time, threading, hashlib, base64

PORT, MODE, TRIG = int(sys.argv[1]), sys.argv[2], sys.argv[3]
USED = set(); LOCK = threading.Lock()

def frame(op, payload=b""): return bytes([0x80 | op, len(payload)]) + payload

class H(http.server.BaseHTTPRequestHandler):
    protocol_version = "HTTP/1.1"
    def log_message(self, *a): pass
    def _json(self, body, code=200):
        b = body.encode(); self.send_response(code); self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(b))); self.end_headers(); self.wfile.write(b)
    def do_POST(self):
        n = int(self.headers.get("Content-Length") or 0); self.rfile.read(n)
        if self.path.endswith("/auth/login"): self._json('{"token":"mock-token"}')
        elif self.path.endswith("/ws/ticket"):
            self._json('{"ticket":"t%d"}' % int(time.time() * 1e6))
        else: self._json("{}", 404)
    def do_GET(self):
        if self.path == "/health": return self._json('{"status":"mock"}')
        if not self.path.startswith("/ws"): return self._json("{}", 404)
        q = self.path.partition("?")[2]
        if self.headers.get("Origin", "").startswith("https://evil"): return self._json("{}", 403)
        if not q.startswith("ticket=") or MODE == "refuse-ws": return self._json("{}", 401)
        t = q[7:]
        with LOCK:
            if t in USED: return self._json("{}", 401)
            USED.add(t)
        key = self.headers["Sec-WebSocket-Key"]
        acc = base64.b64encode(hashlib.sha1((key + "258EAFA5-E914-47DA-95CA-C5AB0DC85B11").encode()).digest()).decode()
        self.wfile.write(("HTTP/1.1 101 Switching Protocols\r\nUpgrade: websocket\r\nConnection: Upgrade\r\n"
                          "Sec-WebSocket-Accept: %s\r\n\r\n" % acc).encode()); self.wfile.flush()
        self.close_connection = True; c = self.connection; c.settimeout(0.3)
        while True:
            try:
                d = c.recv(4096)
                if not d: return
                if (d[0] & 0xF) == 9: c.sendall(frame(10, b"p"))  # pong (payload unmasking not needed for the check)
            except OSError:
                pass
            if os.path.exists(TRIG) and MODE in ("good", "wrong-code"):
                time.sleep(1.0)
                code = 1008 if MODE == "good" else 1000
                c.sendall(frame(8, struct.pack(">H", code) + b"authorization ended")); return

class S(socketserver.ThreadingMixIn, http.server.HTTPServer): daemon_threads = True
S(("127.0.0.1", PORT), H).serve_forever()
