#!/usr/bin/env bash
# LIVE /ws SERVER-SIDE check for SEC-WS-1 F3. PROVISIONAL, NOT RUN against a real server by the author.
# SCOPE (honest): drives the server with raw clients (curl + the stdlib probe ws_probe.py.txt) against a RUNNING
#   catalog-api. It does NOT satisfy rc.1 gate G4's requirement of a live run from catalog-web AND Android through the
#   production nginx edge, and it does NOT cover SEC-WS-1-F1. Those stay OPEN. The audience-filter leg (admin-only
#   events must not reach a non-admin socket) is NOT covered here: it is manual / uncovered and is reported as such.
# Needs: bash, curl, jq, python3. (websocat is NOT needed.) Credentials come from env by NAME only and never appear
#   in argv or logs: BASE=https://host:port ADMIN_USER ADMIN_PASS VICTIM_USER VICTIM_PASS (non-admin test user).
#   Suspension step: set SUSPEND_CMD (a command you supply that suspends VICTIM_USER) or be prompted to do it by hand.
#   Optional: ORIGIN, OUT, F3_BOUND (seconds, default 20), INSECURE=1, SERVER_PID + SERVER_EXE_SHA256 (answering-process
#   identity read on the server host, 11.4.289(D)); without them the run is reported live_proven=NO (UNCONFIRMED).
# Verdict per check: PASS | FAIL | INCONCLUSIVE (INCONCLUSIVE = could not observe; never counted as PASS).
# Overall exit: 0 all PASS, 1 any FAIL, 3 any INCONCLUSIVE and no FAIL, 2 usage/prereq.
set -u
HERE="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
PROBE="${PROBE:-$HERE/ws_probe.py.txt}"
F3_BOUND="${F3_BOUND:-20}"
# deliberately invalid/expired JWT for the ?token= refusal probe; never a real token
EXPIRED_JWT='eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.eyJzdWIiOiJub2JvZHkiLCJleHAiOjF9.AAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAA'

# ---- helpers (unit-testable by sourcing with LIVE_WS_CHECK_SOURCE_ONLY=1) ----
PASSN=0; FAILN=0; INCN=0
record(){ # label verdict detail
  printf '%s | %s | %s\n' "$2" "$1" "$3" >> "${OUT:-/dev/null}/results.txt"
  case "$2" in PASS) PASSN=$((PASSN+1));; FAIL) FAILN=$((FAILN+1));; *) INCN=$((INCN+1));; esac
  printf '%s %s %s\n' "$(date -u +%T)" "$2" "$1 ($3)"
}
check(){ # label observed expected : compares observed vs expected
  local obs="$2" exp="$3"
  if [ -z "$obs" ] || [ "$obs" = "0" ] || [ "$obs" = "000" ]; then
    record "$1" INCONCLUSIVE "no observation (observed='$obs' expect=$exp)"
  elif [ "$obs" = "$exp" ]; then
    record "$1" PASS "observed=$obs expect=$exp"
  else
    record "$1" FAIL "observed=$obs expect=$exp"
  fi
}
hostonly(){ local h="${1#*://}"; h="${h%%/*}"; h="${h%%:*}"; printf '%s' "$h"; }
probe_status(){ # url origin -> HTTP handshake status or empty
  python3 -I "$PROBE" "$1" ${2:+--origin "$2"} --mode handshake 2>/dev/null | tail -1 | jq -r '.handshake // empty' 2>/dev/null
}
login(){
  jq -n --arg u "$1" --arg p "$2" '{username:$u,password:$p}' | \
    curl -sk --max-time 10 -X POST "$BASE/api/v1/auth/login" -H 'Content-Type: application/json' --data @- | \
    jq -r '.token // .access_token // .data.token // empty' 2>/dev/null
}
ticket(){ curl -sk --max-time 10 -X POST "$BASE/api/v1/ws/ticket" -H "Authorization: Bearer $1" | jq -r '.ticket // empty' 2>/dev/null; }
# f3_verdict closecode elapsed bound end -> prints VERDICT|detail
f3_verdict(){
  local code="$1" el="$2" bound="$3" end="$4"
  if [ "$end" != "close" ] || [ -z "$code" ] || [ "$code" = "null" ]; then
    if [ "$end" = "timeout" ]; then
      echo "FAIL|socket still open at end of window (end=$end)"
    else
      echo "INCONCLUSIVE|no close frame observed (end=$end)"
    fi
  elif [ "$code" != "1008" ]; then
    echo "FAIL|close code $code, expected 1008"
  elif awk -v e="$el" 'BEGIN{exit !(e<0)}'; then
    echo "INCONCLUSIVE|closed before the suspension (elapsed=$el)"
  elif awk -v e="$el" -v b="$bound" 'BEGIN{exit !(e>b)}'; then
    echo "FAIL|closed ${el}s after suspension, bound ${bound}s"
  else
    echo "PASS|close 1008 ${el}s after suspension (bound ${bound}s)"
  fi
}
if [ "${LIVE_WS_CHECK_SOURCE_ONLY:-0}" = 1 ]; then return 0 2>/dev/null || exit 0; fi

: "${BASE:?}" "${ADMIN_USER:?}" "${ADMIN_PASS:?}" "${VICTIM_USER:?}" "${VICTIM_PASS:?}"
for t in curl jq python3; do
  command -v "$t" >/dev/null || { echo "prereq missing: $t (INCONCLUSIVE)"; exit 2; }
done
[ -f "$PROBE" ] || { echo "probe missing: $PROBE"; exit 2; }
OUT="${OUT:-live_ws_$(date -u +%Y%m%dT%H%M%SZ)}"; mkdir -p "$OUT"
WS="${BASE/http/ws}"; ORIGIN="${ORIGIN:-$BASE}"; : > "$OUT/results.txt"

# ---- target fields (11.4.289(D)) ----
H="$(hostonly "$BASE")"
{
  echo "base=$BASE"; echo "host_resolved=$(getent hosts "$H" 2>/dev/null | head -1)"; echo "start=$(date -u +%FT%TZ)"
  echo "health_body:"; curl -sk --max-time 10 "$BASE/health"; echo
  echo "health_headers:"; curl -skI --max-time 10 "$BASE/health" | head -20
} > "$OUT/target.txt" 2>&1
if [ -n "${SERVER_PID:-}" ] && [ -n "${SERVER_EXE_SHA256:-}" ]; then
  printf 'pid=%s\nexe_sha256=%s\n' "$SERVER_PID" "$SERVER_EXE_SHA256" > "$OUT/process_identity.txt"; LIVE=yes
else
  echo "MISSING: answering-process identity (pid + executable sha256 read on the server host). Run is UNCONFIRMED / not live-proven (11.4.289(D))." > "$OUT/process_identity.txt"; LIVE=NO
fi

VT=$(login "$VICTIM_USER" "$VICTIM_PASS"); AT=$(login "$ADMIN_USER" "$ADMIN_PASS")
if [ -z "$VT" ] || [ -z "$AT" ]; then
  record "login" INCONCLUSIVE "no token (token field name UNCONFIRMED: adjust jq path in login())"
else
  check "unauthenticated /ws refused"         "$(probe_status "$WS/ws" "$ORIGIN")" 401
  check "invalid JWT in ?token= refused (F2)" "$(probe_status "$WS/ws?token=$EXPIRED_JWT" "$ORIGIN")" 401
  T=$(ticket "$VT")
  if [ -z "$T" ]; then
    record "bad Origin with valid ticket refused" INCONCLUSIVE "no ticket minted"
  else
    check "bad Origin with valid ticket refused" "$(probe_status "$WS/ws?ticket=$T" https://evil.example)" 403
  fi
  T=$(ticket "$VT")
  if [ -z "$T" ]; then
    record "ticket reuse refused (single-use)" INCONCLUSIVE "no ticket minted"
  else
    first="$(probe_status "$WS/ws?ticket=$T" "$ORIGIN")"
    if [ "$first" != "101" ]; then
      record "ticket reuse refused (single-use)" INCONCLUSIVE "first connection did not succeed (status='$first')"
    else
      check "ticket reuse refused (single-use)" "$(probe_status "$WS/ws?ticket=$T" "$ORIGIN")" 401
    fi
  fi

  # ---- F3: connected socket closed (1008) within bound after the victim is suspended ----
  T=$(ticket "$VT")
  READY="$OUT/ready.$$"
  if [ -z "$T" ]; then
    record "F3 socket closed after suspension" INCONCLUSIVE "no ticket minted"
  else
    python3 -I "$PROBE" "$WS/ws?ticket=$T" --origin "$ORIGIN" --mode hold \
      --max-seconds $((F3_BOUND+60)) --ready-file "$READY" > "$OUT/f3_probe.json" 2> "$OUT/f3_probe.err" &
    WP=$!
    for _ in $(seq 1 20); do [ -f "$READY" ] && break; sleep 0.5; done
    if [ ! -f "$READY" ] || ! kill -0 "$WP" 2>/dev/null; then
      record "F3 socket closed after suspension" INCONCLUSIVE "socket not proven alive before suspension (no pong / handshake failed)"
      kill "$WP" 2>/dev/null
    else
      echo "socket alive (pong received). Suspend (or revoke the session of) the victim user now."
      if [ -n "${SUSPEND_CMD:-}" ]; then
        SUSP_EPOCH=$(date +%s.%N); bash -c "$SUSPEND_CMD" > "$OUT/suspend_cmd.out" 2>&1
      else
        printf 'Press Enter right after you suspended the victim: '; read -r _; SUSP_EPOCH=$(date +%s.%N)
      fi
      echo "$SUSP_EPOCH" > "$OUT/suspend_epoch.txt"
      for _ in $(seq 1 $(( (F3_BOUND+5)*2 ))); do kill -0 "$WP" 2>/dev/null || break; sleep 0.5; done
      kill "$WP" 2>/dev/null; wait "$WP" 2>/dev/null
      CC=$(jq -r '.close_code' "$OUT/f3_probe.json" 2>/dev/null); END=$(jq -r '.end' "$OUT/f3_probe.json" 2>/dev/null)
      CE=$(jq -r '.close_epoch // empty' "$OUT/f3_probe.json" 2>/dev/null); RS=$(jq -r '.close_reason' "$OUT/f3_probe.json" 2>/dev/null)
      [ -z "$END" ] && END=timeout
      if [ -n "$CE" ]; then EL=$(awk -v c="$CE" -v s="$SUSP_EPOCH" 'BEGIN{printf "%.2f", c-s}'); else EL=0; fi
      res="$(f3_verdict "$CC" "$EL" "$F3_BOUND" "$END")"
      record "F3 socket closed after suspension (code 1008, reason='$RS')" "${res%%|*}" "${res#*|}"
    fi
  fi
  record "F3 audience filter (admin-only events not delivered to non-admin)" INCONCLUSIVE "NOT COVERED by this script: manual / uncovered"
fi
echo "live_proven=$LIVE (answering-process identity: $([ "$LIVE" = yes ] && echo recorded || echo 'MISSING, UNCONFIRMED'))" | tee -a "$OUT/results.txt"
echo "summary: PASS=$PASSN FAIL=$FAILN INCONCLUSIVE=$INCN" | tee -a "$OUT/results.txt"
( cd "$OUT" && sha256sum ./*.txt ./*.json > SHA256SUMS 2>/dev/null )
[ "$FAILN" -gt 0 ] && exit 1
[ "$INCN" -gt 0 ] && exit 3
exit 0
